Program: Number Guessing Game


import java.util.Scanner;

public class RavenGame {
    public static void main(String[] args) {

        Scanner Ven = new Scanner(System.in);

        int r = 7;
        int a = 0;
        int v = 0;
        int e = 5;
        int n = e;

        System.out.println("==============================");
        System.out.println("     NUMBER GUESSING GAME");
        System.out.println("==============================");

        System.out.println("Guess the number from 1 to 10.");
        System.out.println("You have 5 tries.");

        while (n > 0 && a != r) {

            System.out.print("\nEnter your guess: ");
            a = Ven.nextInt();

            v = v + 1;
            n = n - 1;

            if (a == r) {
                System.out.println("Correct! You win!");
            }
            else if (a < r) {
                System.out.println("Too low. Try again.");
            }
            else {
                System.out.println("Too high. Try again.");
            }

            System.out.println("Tries left: " + n);
        }

        if (a != r) {
            System.out.println("Game over!");
            System.out.println("The secret number was " + r);
        }

        System.out.println("Number of tries used: " + v);

       
    }
}
